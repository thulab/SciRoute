cmake_minimum_required(VERSION 3.16)
project(MultiFileNativeBaseline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(HDF5 1.14.4 EXACT CONFIG REQUIRED COMPONENTS C shared)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(SQLite3 REQUIRED)
find_path(ECCODES_INCLUDE_DIR eccodes.h PATH_SUFFIXES x86_64-linux-gnu REQUIRED)
find_library(ECCODES_LIBRARY eccodes REQUIRED)
if(NOT TARGET hdf5-shared OR NOT HDF5_ENABLE_THREADSAFE)
  message(FATAL_ERROR "Require the same thread-safe HDF5 1.14.4 as the daemon")
endif()
add_executable(multifile_native multifile_native.cpp)
target_include_directories(multifile_native PRIVATE ${ECCODES_INCLUDE_DIR})
target_link_libraries(multifile_native PRIVATE hdf5-shared OpenSSL::Crypto SQLite::SQLite3 ${ECCODES_LIBRARY})
